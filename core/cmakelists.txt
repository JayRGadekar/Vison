find_package(Threads REQUIRED)

add_library(vison_core STATIC
    src/vison.cpp
    src/device.cpp
    src/queue.cpp
    src/download.cpp
    src/hf_link.cpp
)

target_include_directories(vison_core PUBLIC include)
# ggml for real device enumeration in device.cpp. Only the generic
# ggml-backend API is used (dev_count/description/memory), so this works for
# whichever backends are actually built - it does not tie core to Vulkan.
target_link_libraries(vison_core PUBLIC Threads::Threads ggml)

if(WIN32)
    # Native HTTPS downloads (replaces the old python scripts/download.py shell-out).
    target_link_libraries(vison_core PUBLIC winhttp)
endif()

if(VISON_CUDA)
    target_compile_definitions(vison_core PUBLIC VISON_CUDA)
endif()

if(VISON_VULKAN)
    target_compile_definitions(vison_core PUBLIC VISON_VULKAN)
endif()

# Pure-logic tests (no network, no GPU). Run: ctest -C Release -R hf_link
add_executable(hf_link_test tests/hf_link_test.cpp)
target_link_libraries(hf_link_test PRIVATE vison_core)
enable_testing()
add_test(NAME hf_link COMMAND hf_link_test)
